  86 | async function parseBody<T extends z.ZodType>(
  87 | 	request: Request,
  88 | 	schema: T,
  89 | ): Promise<z.infer<T>> {
  90 | 	let raw: unknown;
  91 | 	try {
  92 | 		raw = await request.json();
  93 | 	} catch {
  94 | 		throw new Problem(400, "Bad Request", "The body is not valid JSON.");
  95 | 	}
  96 | 	const result = schema.safeParse(raw);
  97 | 	if (!result.success) {
  98 | 		throw new Problem(400, "Bad Request", "The body is invalid.", {
  99 | 			issues: result.error.issues.map((issue) => ({
 100 | 				path: issue.path,
 101 | 				code: issue.code,
 102 | 				message: issue.message,
 103 | 			})),
 104 | 		});
 105 | 	}
 106 | 	return result.data;
 107 | }

 147 | // ---------------------------------------------------------------------------
 148 | // Tokens (HS256, secret from JWT_SECRET)
 149 | // ---------------------------------------------------------------------------
 150 | 
 151 | const ISSUER = "ticketing";
 152 | const TOKEN_SECONDS = 15 * 60;
 153 | 
 154 | function secret(): string {
 155 | 	const value = process.env.JWT_SECRET;
 156 | 	if (!value || value.length < 32) {
 157 | 		throw new Error("JWT_SECRET must be set, with at least 32 characters.");
 158 | 	}
 159 | 	return value;
 160 | }

 162 | async function issueToken(organizer: Organizer): Promise<string> {
 163 | 	const now = Math.floor(Date.now() / 1000);
 164 | 	return sign(
 165 | 		{ sub: organizer.id, iss: ISSUER, iat: now, exp: now + TOKEN_SECONDS },
 166 | 		secret(),
 167 | 		"HS256",
 168 | 	);
 169 | }
 170 | 
 171 | /** The signed-in organizer, from `Authorization: Bearer <token>`, or a 401. */
 172 | async function caller(request: Request): Promise<Organizer> {
 173 | 	const header = request.headers.get("authorization") ?? "";
 174 | 	const match = /^Bearer (.+)$/i.exec(header);
 175 | 	const unauthenticated = new Problem(
 176 | 		401,
 177 | 		"Unauthorized",
 178 | 		"A valid bearer token is required.",
 179 | 	);
 180 | 	if (!match?.[1]) throw unauthenticated;
 181 | 	let subject: unknown;
 182 | 	try {
 183 | 		const payload = await verify(match[1], secret(), "HS256");
 184 | 		if (payload.iss !== ISSUER) throw unauthenticated;
 185 | 		subject = payload.sub;
 186 | 	} catch {
 187 | 		throw unauthenticated;
 188 | 	}
 189 | 	const organizer =
 190 | 		typeof subject === "string" ? organizers.get(subject) : undefined;
 191 | 	if (!organizer) throw unauthenticated;
 192 | 	return organizer;
 193 | }
 194 | 
 195 | /** The event of `id` if `organizer` owns it; another organizer's is a 404, as if it did not exist. */
 196 | function ownedEvent(id: string, organizer: Organizer): Event {
 197 | 	const event = events.get(id);
 198 | 	if (!event || event.organizerId !== organizer.id) {
 199 | 		throw new Problem(404, "Not Found", "No such event.");
 200 | 	}
 201 | 	return event;
 202 | }

 216 | 			"Conflict",
 217 | 			"An organizer with this email already exists.",
 218 | 		);

 285 | app.post("/events/:eventId/tickets", async (c) => {
 286 | 	const event = events.get(c.req.param("eventId"));
 287 | 	if (!event) throw new Problem(404, "Not Found", "No such event.");
 288 | 	const body = await parseBody(c.req.raw, BuyTicketBody);
 289 | 	const sold = ticketCount.get(event.id) ?? 0;
 290 | 	if (sold >= event.capacity) {

 292 | 	}
 293 | 	const ticket: Ticket = {
 294 | 		id: crypto.randomUUID(),
 295 | 		eventId: event.id,
 296 | 		buyerEmail: body.buyerEmail,
 297 | 		createdAt: new Date().toISOString(),
 298 | 	};
 299 | 	tickets.set(ticket.id, ticket);
 300 | 	ticketCount.set(event.id, sold + 1);
 301 | 	return c.json({ ticket }, 201);
 302 | });
